cmake_minimum_required(VERSION 3.22)
project(appearance_startup LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(LingmoSDK REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
add_executable(appearance-startup main.cpp ../../src/ui/appearance.cpp)
target_include_directories(appearance-startup PRIVATE ../../src)
target_link_libraries(appearance-startup PRIVATE Lingmo::SDK PkgConfig::SYSTEMD)
